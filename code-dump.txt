            function mountApp(
                navigator,
                {
                    host,
                    tabBar,
                    transition = { duration: 200 },
                    history = false,
                    swipe = false,
                    arrows = null
                }
            ) {
                const hostEl =
                    typeof host === "string"
                        ? document.querySelector(host)
                        : host;
                if (!hostEl)
                    throw new Error(`mountApp: no element for "${host}"`);

                // ScreenHost first — otherwise deep-link restores flash the
                // initial screen before flipping to the restored one.
                const screenHost = new ScreenHost(navigator, { transition });
                screenHost.mount(hostEl);

                const swipeInstance = swipe
                    ? new SwipeNavigation(navigator).mount(hostEl)
                    : null;

                const tabBarEl = tabBar
                    ? typeof tabBar === "string"
                        ? document.querySelector(tabBar)
                        : tabBar
                    : null;
                const tabBarInstance = tabBarEl
                    ? new TabBar(navigator).mount(tabBarEl)
                    : null;

                const historyInstance = history
                    ? new HistorySync(navigator, history).start()
                    : null;

                if (arrows) {
                    document
                        .querySelector(arrows.back)
                        ?.addEventListener("click", () => navigator.back());
                    document
                        .querySelector(arrows.forward)
                        ?.addEventListener("click", () => navigator.forward());
                }

                return {
                    screenHost,
                    tabBar: tabBarInstance,
                    swipe: swipeInstance,
                    history: historyInstance
                };
            }

export default class TabBar {
    constructor(navigation) {
        if (
            !navigation ||
            typeof navigation.selectTab !== "function" ||
            typeof navigation.on !== "function" ||
            typeof navigation.off !== "function"
        ) {
            throw new TypeError("TabBar requires a TabNavigator.");
        }

        this.navigation = navigation;
        this.container = null;
        this.tabList = null;
        this.buttons = new Map();
        this._listener = () => this._updateSelection();
    }

    get isMounted() {
        return this.container !== null;
    }

    get element() {
        return this.tabList;
    }

    mount(target) {
        if (!(target instanceof Element)) {
            throw new TypeError("TabBar requires a valid DOM container.");
        }

        if (this.container) {
            throw new Error("TabBar is already mounted. Unmount it first.");
        }

        const tabs = this.navigation.tabs;

        if (!Array.isArray(tabs) || tabs.length === 0) {
            throw new Error("TabBar requires at least one configured tab.");
        }

        const nav = document.createElement("div");
        nav.className = "tab-bar";
        nav.setAttribute("role", "tablist");
        nav.setAttribute("aria-label", "Main navigation");

        const nextButtons = new Map();

        for (const tab of tabs) {
            const button = document.createElement("button");
            button.type = "button";
            button.className = "tab-bar__button";
            button.setAttribute("role", "tab");
            button.setAttribute("aria-label", tab.label);
            button.setAttribute("aria-selected", "false");
            button.setAttribute("data-tab-id", tab.id);
            button.tabIndex = -1;

            if (tab.icon) {
                const icon = document.createElement("span");
                icon.className = "tab-bar__icon";
                icon.setAttribute("aria-hidden", "true");
                icon.textContent = tab.icon;
                button.appendChild(icon);
            }

            const label = document.createElement("span");
            label.className = "tab-bar__label";
            label.textContent = tab.label;
            button.appendChild(label);

            button.addEventListener("click", () =>
                this._handleTabClick(tab.id)
            );

            nextButtons.set(tab.id, button);
            nav.appendChild(button);
        }

        target.replaceChildren(nav);

        this.container = target;
        this.tabList = nav;

        for (const [id, button] of nextButtons) {
            this.buttons.set(id, button);
        }

        this.navigation.on("change", this._listener);
        this._updateSelection();

        return this;
    }

    unmount() {
        if (!this.container) {
            return this;
        }

        this.navigation.off("change", this._listener);

        this.container.replaceChildren();

        this.container = null;
        this.tabList = null;
        this.buttons.clear();

        return this;
    }

    _updateSelection() {
        if (!this.container) {
            return;
        }

        const selectedTabId = this.navigation.selectedTabId;

        for (const [id, button] of this.buttons) {
            const selected = id === selectedTabId;

            button.setAttribute("aria-selected", String(selected));
            button.classList.toggle("is-selected", selected);
            button.tabIndex = selected ? 0 : -1;
        }
    }

    _handleTabClick(id) {
        this.navigation.selectTab(id);
        this._updateSelection();
    }
}

    "size",
    "number",
    "enum",
    "string",
    "boolean",
    "function",
    "tagName",
    "fontWeight",
    "style",

function isPlainObject(value) {
    if (value === null || typeof value !== "object") {
        return false;
    }

    const prototype = Object.getPrototypeOf(value);

    return prototype === Object.prototype || prototype === null;
}

/**
 * Clone and freeze schema data recursively.
 * Functions are retained as-is so custom validators remain callable.
 */

function deepFreeze(value, seen = new WeakMap()) {
    if (
        value === null ||
        (typeof value !== "object" && typeof value !== "function")
    ) {
        return value;
    }

    if (typeof value === "function") {
        return value;
    }

    if (seen.has(value)) {
        return seen.get(value);
    }

    const clone = Array.isArray(value)
        ? []
        : Object.create(Object.getPrototypeOf(value));

    seen.set(value, clone);

    for (const key of Reflect.ownKeys(value)) {
        const descriptor = Object.getOwnPropertyDescriptor(value, key);

        if (descriptor && "value" in descriptor) {
            descriptor.value = deepFreeze(descriptor.value, seen);
        }

        Object.defineProperty(clone, key, descriptor);
    }

    return Object.freeze(clone);
}

function validateSchema(type, schema) {
    if (typeof type !== "string" || !/^[a-z][a-zA-Z0-9-]*$/.test(type)) {
        throw new TypeError(`Invalid component schema name "${type}".`);
    }

    if (!isPlainObject(schema)) {
        throw new TypeError(`Schema "${type}" must be a plain object.`);
    }

    const allowedSchemaKeys = new Set(["properties", "defaults", "required"]);

    for (const key of Object.keys(schema)) {
        if (!allowedSchemaKeys.has(key)) {
            throw new TypeError(
                `Unknown schema setting "${key}" in "${type}".`
            );
        }
    }

    const properties = schema.properties ?? {};
    const defaults = schema.defaults ?? {};
    const required = schema.required ?? [];

    if (!isPlainObject(properties)) {
        throw new TypeError(
            `Schema "${type}" properties must be a plain object.`
        );
    }

    if (!isPlainObject(defaults)) {
        throw new TypeError(
            `Schema "${type}" defaults must be a plain object.`
        );
    }

    if (
        !Array.isArray(required) ||
        !required.every(name => typeof name === "string") ||
        new Set(required).size !== required.length
    ) {
        throw new TypeError(
            `Schema "${type}" required must be an array of unique property names.`
        );
    }

    const propertyRules = {};

    for (const [name, rule] of Object.entries(properties)) {
        if (!/^[a-zA-Z][a-zA-Z0-9_-]*$/.test(name)) {
            throw new TypeError(
                `Invalid property name "${name}" in schema "${type}".`
            );
        }

        if (!isPlainObject(rule)) {
            throw new TypeError(
                `Schema "${type}" property "${name}" must have a rule object.`
            );
        }

        if (!VALID_SCHEMA_KINDS.has(rule.kind)) {
            throw new TypeError(
                `Unsupported rule kind "${rule.kind}" for "${type}.${name}".`
            );
        }

        const allowedRuleKeys = new Set([
            "kind",
            "values",
            "validate",
            "message"
        ]);

        for (const key of Object.keys(rule)) {
            if (!allowedRuleKeys.has(key)) {
                throw new TypeError(
                    `Unknown rule setting "${key}" for "${type}.${name}".`
                );
            }
        }

        if (
            rule.kind === "enum" &&
            (!Array.isArray(rule.values) || rule.values.length === 0)
        ) {
            throw new TypeError(
                `Enum "${type}.${name}" must declare at least one allowed value.`
            );
        }

        if (
            rule.kind === "enum" &&
            rule.values.some(
                value =>
                    value !== null &&
                    !["string", "number", "boolean"].includes(typeof value)
            )
        ) {
            throw new TypeError(
                `Enum "${type}.${name}" may contain only primitive values.`
            );
        }

        if (
            rule.validate !== undefined &&
            typeof rule.validate !== "function"
        ) {
            throw new TypeError(
                `Custom validator for "${type}.${name}" must be a function.`
            );
        }

        if (rule.message !== undefined && typeof rule.message !== "string") {
            throw new TypeError(
                `Validation message for "${type}.${name}" must be a string.`
            );
        }

        propertyRules[name] = rule;
    }

    for (const name of required) {
        if (
            !Object.hasOwn(propertyRules, name) &&
            !Object.hasOwn(COMMON_PROPERTIES, name)
        ) {
            throw new TypeError(
                `Required property "${type}.${name}" has no declared property rule.`
            );
        }
    }

    for (const name of Object.keys(defaults)) {
        if (
            !Object.hasOwn(propertyRules, name) &&
            !Object.hasOwn(COMMON_PROPERTIES, name)
        ) {
            throw new TypeError(
                `Default "${type}.${name}" has no declared property rule.`
            );
        }
    }

    return deepFreeze({
        properties: propertyRules,
        defaults: { ...defaults },
        required: [...required]
    });
}

/**
 * Register a component schema.
 *
 * Existing names are protected unless { replace: true } is supplied.
 * Component implementations and renderer support must be added separately.
 */
export function registerComponentSchema(type, schema, options = {}) {
    if (!isPlainObject(options)) {
        throw new TypeError(
            "Schema registration options must be a plain object."
        );
    }

    for (const key of Object.keys(options)) {
        if (key !== "replace") {
            throw new TypeError(`Unknown registration option "${key}".`);
        }
    }

    if (options.replace !== undefined && typeof options.replace !== "boolean") {
        throw new TypeError('Registration option "replace" must be a boolean.');
    }

    if (registry.has(type) && options.replace !== true) {
        throw new Error(`Component schema "${type}" is already registered.`);
    }

    const normalized = validateSchema(type, schema);

    registry.set(type, normalized);

    return getComponentSchema(type);
}

export function getComponentSchema(type) {
    return registry.get(type) ?? null;
}

export function hasComponentSchema(type) {
    return registry.has(type);
}

export function listComponentTypes() {
    return [...registry.keys()];
}

function validateSize(name, value) {
    if (value == null) {
        return;
    }

    const valid =
        (typeof value === "number" && Number.isFinite(value) && value >= 0) ||
        (typeof value === "string" && value.trim().length > 0);

    for (const name of SIZE_PROPERTIES) {
        const value = props[name];

        if (value != null) {

        }
    }

    if (props.align != null) {
        if (!ALIGN_VALUES.has(props.align)) {
            throw new TypeError(

            );

        layout.align = props.align;
    }

    if (props.justify != null) {
        if (!JUSTIFY_VALUES.has(props.justify)) {
            throw new TypeError(
                `Invalid layout justify value: "${props.justify}".`
            );
        }

        layout.justify = props.justify;
    }

    return layout;
}

function normalizeNode(node) {
    if (typeof node === "string" || typeof node === "number") {
        return node;
    }

    if (Array.isArray(node)) {
        return {
            type: "fragment",
            props: {},
            children: node.map(normalizeNode)
        };
    }

    const schema = getComponentSchema(node.type);

    if (!schema) {
        throw new TypeError(
            `Unknown component type "${node.type}".`
        );
    }

    const props = {
        ...schema.defaults,
        ...node.props
    };

    const children = (node.children ?? []).map(normalizeNode);

    return {
        type: node.type,
        props,
        children
    };
}

export default function normalizeComponent(component) {
    const result = validateDefinition(component);

    if (!result.valid) {
        throw new TypeError(
            `UI definition validation failed:\n- ${result.errors.join("\n- ")}`
        );
    }

    return normalizeNode(component);
}

import { COMMON_PROPERTIES, getComponentSchema } from "./componentSchemas.js";

function isPlainObject(value) {
    if (value === null || typeof value !== "object") {
        return false;
    }

    const prototype = Object.getPrototypeOf(value);

    return prototype === Object.prototype || prototype === null;
}

function isValidSize(value) {
    if (typeof value === "number") {
        return Number.isFinite(value) && value >= 0;
    }

    if (typeof value !== "string") {
        return false;
    }

    const text = value.trim();

    return (
        text.length > 0 &&
        !/[;{}]/.test(text) &&
        !/[\u0000-\u0008\u000B\u000C\u000E-\u001F]/.test(text)
    );
}

function isValidTagName(value) {
    return typeof value === "string" && /^[a-zA-Z][a-zA-Z0-9-]*$/.test(value);
}

function isValidFontWeight(value) {
    if (typeof value === "number") {
        return Number.isFinite(value) && value >= 1 && value <= 1000;
    }

    return (
        typeof value === "string" &&
        /^(normal|bold|bolder|lighter|[1-9]00)$/.test(value)
    );
}

function validateStyle(value, errors, path) {
    if (!isPlainObject(value)) {
        errors.push(`${path}: "style" must be a plain object.`);
        return;
    }

    for (const [name, entry] of Object.entries(value)) {
        if (!/^(--[a-zA-Z0-9_-]+|[a-zA-Z][a-zA-Z0-9]*)$/.test(name)) {
            errors.push(`${path}: Invalid CSS property name "${name}".`);
            continue;
        }

        const validEntry =
            entry == null ||
            typeof entry === "string" ||
            (typeof entry === "number" && Number.isFinite(entry));

        if (!validEntry) {
            errors.push(
                `${path}: CSS property "${name}" must be a string, finite number, or null.`
            );
        }
    }
}

function validateAttributes(value, errors, path) {
    if (!isPlainObject(value)) {
        errors.push(`${path}: "attributes" must be a plain object.`);
        return;
    }

    for (const [name, entry] of Object.entries(value)) {
        if (!/^[a-zA-Z_:][a-zA-Z0-9:._-]*$/.test(name)) {
            errors.push(`${path}: Invalid HTML attribute name "${name}".`);
            continue;
        }

        const validEntry =
            entry == null ||
            typeof entry === "string" ||
            typeof entry === "number" ||
            typeof entry === "boolean";

        if (!validEntry) {
            errors.push(
                `${path}: Attribute "${name}" must be a string, number, boolean, or null.`
            );
        }
    }
}

function validateProperty(name, value, rule, props, errors, path) {
    let valid = true;

    switch (rule.kind) {
        case "size":
            valid = isValidSize(value);
            break;

        case "number":
            valid = typeof value === "number" && Number.isFinite(value);
            break;

        case "enum":
            valid = rule.values.includes(value);
            break;

        case "string":
            valid = typeof value === "string";
            break;

        case "boolean":
            valid = typeof value === "boolean";
            break;

        case "function":
            valid = typeof value === "function";
            break;

        case "tagName":
            valid = isValidTagName(value);
            break;

        case "fontWeight":
            valid = isValidFontWeight(value);
            break;

        case "style":
            validateStyle(value, errors, path);
            break;

        case "attributes":
            validateAttributes(value, errors, path);
            break;

        default:
            errors.push(
                `${path}: Unknown schema kind "${rule.kind}" for "${name}".`
            );
            return;
    }

    if (!valid) {
        errors.push(`${path}: Invalid value for property "${name}".`);
        return;
    }

    if (typeof rule.validate === "function") {
        try {
            const result = rule.validate(value, props);

            if (result !== true) {
                const message =
                    typeof result === "string"
                        ? result
                        : (rule.message ??
                          `Custom validation failed for property "${name}".`);

                errors.push(`${path}: ${message}`);
            }
        } catch (error) {
            errors.push(
                `${path}: Custom validator for "${name}" threw an error: ${error.message}`
            );
        }
    }
}

export default function validateComponent(type, props = {}, path = "root") {
    const errors = [];
    const componentSchema = getComponentSchema(type);

    if (!componentSchema) {
        return [`${path}: Unknown component type "${type}".`];
    }

    if (!isPlainObject(props)) {
        return [`${path}: Component props must be a plain object.`];
    }

    const properties = {
        ...COMMON_PROPERTIES,
        ...componentSchema.properties
    };

    for (const name of componentSchema.required) {
        if (!Object.hasOwn(props, name) || props[name] === undefined) {
            errors.push(
                `${path}: Required property "${name}" is missing for component type "${type}".`
            );
        }
    }

    for (const [name, value] of Object.entries(props)) {
        const rule = properties[name];

        if (!rule) {
            errors.push(
                `${path}: Unsupported property "${name}" for component type "${type}".`
            );
            continue;
        }

        // Optional values may be omitted or explicitly set to undefined.
        if (value === undefined) {
            continue;
        }

        validateProperty(name, value, rule, props, errors, path);
    }

    return errors;
}

import validateComponent from "./validateComponent.js";

function isPlainObject(value) {
    if (value === null || typeof value !== "object") {
        return false;
    }

    const prototype = Object.getPrototypeOf(value);

    return prototype === Object.prototype || prototype === null;
}

export default function validateDefinition(definition) {
    const errors = [];
    const activePath = new WeakSet();

    function visit(node, path) {
        if (typeof node === "string") {
            return;
        }

        if (typeof node === "number") {
            if (!Number.isFinite(node)) {
                errors.push(
                    `${path}: Numeric children must be finite numbers.`
                );
            }

            return;
        }

        if (node === null || typeof node !== "object") {
            errors.push(`${path}: Invalid component or child.`);
            return;
        }

        if (activePath.has(node)) {
            errors.push(
                `${path}: Circular component references are not allowed.`
            );
            return;
        }

        activePath.add(node);

        try {
            if (Array.isArray(node)) {
                node.forEach((child, index) => {
                    visit(child, `${path}[${index}]`);
                });

                return;
            }

            if (
                !isPlainObject(node) &&
                !(
                    typeof node.type === "string" &&
                    "props" in node &&
                    "children" in node
                )
            ) {
                errors.push(
                    `${path}: Expected a plain-object definition or component instance.`
                );

                return;
            }

            if (typeof node.type !== "string") {
                errors.push(`${path}: Component type must be a string.`);

                return;
            }

            const props = node.props ?? {};
            const children = node.children ?? [];

            errors.push(...validateComponent(node.type, props, path));

            if (!Array.isArray(children)) {
                errors.push(`${path}: Component children must be an array.`);

                return;
            }

            children.forEach((child, index) => {
                visit(child, `${path}.children[${index}]`);
            });
        } finally {
            activePath.delete(node);
        }
    }

    // A root must be a component definition or an array of nodes.
    if (definition === null || typeof definition !== "object") {
        return {
            valid: false,
            errors: ["root: The UI root must be a component or array."]
        };
    }

    visit(definition, "root");

    return {
        valid: errors.length === 0,
        errors
    };
}

const STATE_VERSION = 1;
const VALID_MODES = Object.freeze(["hash", "history"]);

function toSlug(name) {
    return String(name)
        .replace(/([a-z0-9])([A-Z])/g, "$1-$2")
        .replace(/_/g, "-")
        .toLowerCase();
}

function isPlainObject(value) {
    if (value === null || typeof value !== "object") return false;
    const prototype = Object.getPrototypeOf(value);
    return prototype === Object.prototype || prototype === null;
}

export default class HistorySync {
    constructor(navigator, options = {}) {
        if (
            !navigator ||
            typeof navigator.on !== "function" ||
            typeof navigator.off !== "function" ||
            typeof navigator.restore !== "function"
        ) {
            throw new TypeError(
                "HistorySync requires a TabNavigator with restore()."
            );
        }

        if (!isPlainObject(options)) {
            throw new TypeError("HistorySync options must be a plain object.");
        }

        for (const key of Object.keys(options)) {
            if (key !== "mode" && key !== "basePath") {
                throw new TypeError(`Unknown HistorySync option "${key}".`);
            }
        }

        const mode = options.mode ?? "hash";
        if (!VALID_MODES.includes(mode)) {
            throw new TypeError(
                `HistorySync mode must be one of: ${VALID_MODES.join(", ")}.`
            );
        }

        const basePath = options.basePath ?? "";
        if (typeof basePath !== "string") {
            throw new TypeError("HistorySync basePath must be a string.");
        }

        this.navigator = navigator;
        this.mode = mode;
        this.basePath = basePath.replace(/\/+$/, "");

        this._started = false;
        this._suppress = false;
        this._pendingPop = false;
        this._historyPointer = 0;
        this._slugToName = this._buildSlugMap();
        this._lastOp = "init";

        this._onChange = event => this._handleChange(event);
        this._onPopState = event => this._handlePopState(event);
    }

    get isStarted() {
        return this._started;
    }

    start() {
        if (this._started) return this;

        this._historyPointer = 0;
        this._pendingPop = false;
        this._lastOp = "init";

        this._restoreOnStart();
        this._writeState({ replace: true });

        this.navigator.on("change", this._onChange);
        window.addEventListener("popstate", this._onPopState);
        this._started = true;

        return this;
    }

    stop() {
        if (!this._started) return this;

        this.navigator.off("change", this._onChange);
        window.removeEventListener("popstate", this._onPopState);

        this._started = false;
        this._pendingPop = false;

        return this;
    }

    _buildSlugMap() {
        const map = new Map();
        const registry = this.navigator.registry;

        if (!registry || typeof registry.list !== "function") {
            return map;
        }

        for (const name of registry.list()) {
            const slug = toSlug(name);

            if (map.has(slug) && map.get(slug) !== name) {
                console.warn(
                    `HistorySync: screen names "${map.get(slug)}" and "${name}" both slug to "${slug}". ` +
                        "Deep links to this screen will be ambiguous."
                );
            }

            map.set(slug, name);
        }

        return map;
    }

    _handleChange(event) {
        if (this._suppress) return;

        if (event.action === "back") {
            this._handleAppBack();
            return;
        }

        if (event.action === "forward") {
            this._handleAppForward();
            return;
        }

        const replace = event.action === "selectTab";
        this._writeState({ replace });
    }

    _handleAppBack() {
        // Defensive: if a previous history.back() never fired popstate,
        // prefer replaceState over compounding broken state.
        if (this._pendingPop) {
            this._pendingPop = false;
            this._writeState({ replace: true });
            return;
        }

        if (this._historyPointer > 0) {
            this._historyPointer -= 1;
            this._pendingPop = true;
            this._lastOp = "back";
            history.back();
        } else {
            // No prior app entry (fresh deep-link paste, or start-of-session).
            // Replacing keeps browser history honest at the cost of the
            // pasted entry — the correct trade-off.
            this._writeState({ replace: true });
        }
    }

    _handleAppForward() {
        if (this._pendingPop) {
            // Previous forward() never received its popstate; treat the
            // app as already in the target state.
            this._pendingPop = false;
            this._writeState({ replace: true });
            return;
        }

        if (this._lastOp === "pop") {
            this._pendingPop = true;
            this._historyPointer += 1;
            this._lastOp = "forward";
            history.forward();
        } else {
            this._writeState({ replace: true });
        }
    }

    _handlePopState(event) {
        if (this._pendingPop) {
            this._pendingPop = false;
            this._lastOp = "pop";
            return;
        }

        this._lastOp = "pop"; // ← new

        this._historyPointer =
            typeof event.state?._pointer === "number"
                ? event.state._pointer
                : 0;

        this._suppress = true;

        try {
            const state = event.state;

            if (
                state?.version === STATE_VERSION &&
                state.tabId &&
                Array.isArray(state.stack) &&
                state.stack.length > 0
            ) {
                this.navigator.restore(state.tabId, state.stack);
                return;
            }

            const parsed = this._parseLocation();
            if (parsed) {
                this.navigator.restore(parsed.tabId, parsed.entries);
            }
        } catch (error) {
            console.error("HistorySync: popstate restore failed", error);
        } finally {
            this._suppress = false;
        }
    }

    _restoreOnStart() {
        if (history.state?.version === STATE_VERSION) {
            if (typeof history.state._pointer === "number") {
                this._historyPointer = history.state._pointer;
            }

            try {
                this.navigator.restore(
                    history.state.tabId,
                    history.state.stack
                );
                return true;
            } catch (error) {
                console.error(
                    "HistorySync: restore from state failed, falling back to URL",
                    error
                );
            }
        }

        const parsed = this._parseLocation();

        if (parsed) {
            try {
                this.navigator.restore(parsed.tabId, parsed.entries);
                return true;
            } catch (error) {
                console.error("HistorySync: restore from URL failed", error);
            }
        }

        return false;
    }

    _parseLocation() {
        let route;

        if (this.mode === "hash") {
            route = location.hash.slice(1);
            if (!route || !route.startsWith("/")) return null;
        } else {
            const path = location.pathname;
            if (this.basePath) {
                if (!path.startsWith(this.basePath)) return null;
                route = path.slice(this.basePath.length) || "/";
            } else {
                route = path;
            }
            route += location.search;
        }

        const [pathPart, queryPart = ""] = route.split("?");
        const segments = pathPart.split("/").filter(Boolean);
        if (segments.length === 0) return null;

        const tabId = this._findTabBySlug(segments[0]);
        if (!tabId) return null;

        const tab = this.navigator.tabs.find(entry => entry.id === tabId);
        if (!tab) return null;

        const entries = [{ name: tab.screen, params: {} }];

        for (let i = 1; i < segments.length; i++) {
            const name = this._slugToName.get(segments[i]);
            if (!name) return null;
            entries.push({ name, params: {} });
        }

        const params = Object.fromEntries(new URLSearchParams(queryPart));
        entries[entries.length - 1].params = params;

        return { tabId, entries };
    }

    _findTabBySlug(slug) {
        for (const tab of this.navigator.tabs) {
            if (toSlug(tab.id) === slug) return tab.id;
        }
        return null;
    }

    _writeState({ replace }) {
        const current = this.navigator.current;
        if (!current) return;

        const tabId = current.tabId ?? this.navigator.selectedTabId;
        if (!tabId) return;

        const stack = this.navigator.stack;
        if (!Array.isArray(stack) || stack.length === 0) return;

        if (!replace) {
            this._historyPointer += 1;
            this._lastOp = "push";
        }

        const route = this._buildRoute(tabId, stack);
        const url = this._buildUrl(route);

        const state = {
            version: STATE_VERSION,
            _pointer: this._historyPointer,
            tabId,
            stack: stack.map(entry => ({
                name: entry.name,
                params: entry.params ?? {}
            }))
        };

        const method = replace ? "replaceState" : "pushState";

        try {
            history[method](state, "", url);
        } catch (error) {
            console.error("HistorySync: failed to write URL", error);
        }
    }

    _buildRoute(tabId, stack) {
        const tabSlug = toSlug(tabId);
        const rest = stack.slice(1).map(entry => toSlug(entry.name));
        const path = "/" + [tabSlug, ...rest].join("/");

        const topParams = stack[stack.length - 1].params ?? {};
        const query = new URLSearchParams(topParams).toString();

        return query ? `${path}?${query}` : path;
    }

    _buildUrl(route) {
        if (this.mode === "hash") {
            return `${location.pathname}${location.search}#${route}`;
        }
        return `${this.basePath}${route}`;
    }
}

import ScreenBase from "./ScreenBase.js";

const SCREEN_DEFINITION = Symbol("screen-definition");

function isPlainObject(value) {
    if (value === null || typeof value !== "object") {
        return false;
    }

    const prototype = Object.getPrototypeOf(value);

    return (
        prototype === Object.prototype ||
        prototype === null
    );
}

function isClassConstructor(value) {
    return /^class\s/.test(
        Function.prototype.toString.call(value)
    );
}

export function isScreenDefinition(value) {
    return (
        value !== null &&
        typeof value === "object" &&
        value[SCREEN_DEFINITION] === true
    );
}

export default function Screen(renderOrClass, options = {}) {
    if (typeof renderOrClass !== "function") {
        throw new TypeError(
            "Screen requires a render function or a ScreenBase subclass."
        );
    }

    if (!isPlainObject(options)) {
        throw new TypeError(
            "Screen options must be a plain object."
        );
    }

    for (const key of Object.keys(options)) {
        if (key !== "preserveState") {
            throw new TypeError(
                `Unknown screen option "${key}".`
            );
        }
    }

    if (
        options.preserveState !== undefined &&
        typeof options.preserveState !== "boolean"
    ) {
        throw new TypeError(
            'Screen option "preserveState" must be a boolean.'
        );
    }

    const isScreenClass =
        renderOrClass.prototype instanceof ScreenBase;

    if (
        isClassConstructor(renderOrClass) &&
        !isScreenClass
    ) {
        throw new TypeError(
            "Class-based screens must extend ScreenBase."
        );
    }

    if (
        isScreenClass &&
        renderOrClass.prototype.render === ScreenBase.prototype.render
    ) {
        throw new TypeError(
            "A ScreenBase subclass must implement render()."
        );
    }

    const definition = {
        kind: isScreenClass ? "class" : "factory",

        render: isScreenClass ? null : renderOrClass,

        ScreenClass: isScreenClass ? renderOrClass : null,

        options: Object.freeze({
            preserveState: options.preserveState ?? true
        })
    };

    Object.defineProperty(definition, SCREEN_DEFINITION, {
        value: true
    });

    return Object.freeze(definition);
}

import DOMRenderer from "../renderers/dom/DOMRenderer.js";
import ScreenTransition from "./ScreenTransition.js";

function isPlainObject(value) {
    if (value === null || typeof value !== "object") {
        return false;
    }
    const prototype = Object.getPrototypeOf(value);
    return prototype === Object.prototype || prototype === null;
}

function directionFor(action) {
    if (action === "navigate") return "forward";
    if (action === "back") return "backward";
    return "fade";
}

export default class ScreenHost {
    constructor(navigation, renderer, options = {}) {
        if (isPlainObject(renderer)) {
            [renderer, options] = [new DOMRenderer(), renderer];
        }

        if (renderer === undefined || renderer === null) {
            renderer = new DOMRenderer();
        }

        if (
            !navigation ||
            typeof navigation.on !== "function" ||
            typeof navigation.off !== "function"
        ) {
            throw new TypeError(
                "ScreenHost requires a ScreenManager or TabNavigator."
            );
        }

        if (
            !renderer ||
            typeof renderer.render !== "function" ||
            typeof renderer.renderToElement !== "function"
        ) {
            throw new TypeError(
                "ScreenHost requires a renderer with render() and renderToElement() methods."
            );
        }

        if (!isPlainObject(options)) {
            throw new TypeError("ScreenHost options must be a plain object.");
        }

        for (const key of Object.keys(options)) {
            if (key !== "transition" && key !== "preserveScroll") {
                throw new TypeError(`Unknown ScreenHost option "${key}".`);
            }
        }

        this.preserveScroll = options.preserveScroll ?? true;

        if (typeof this.preserveScroll !== "boolean") {
            throw new TypeError(
                'ScreenHost option "preserveScroll" must be a boolean.'
            );
        }

        this.navigation = navigation;
        this.renderer = renderer;

        const transitionOption = options.transition;

        if (transitionOption === false) {
            this.transition = null;
        } else if (
            transitionOption === undefined ||
            transitionOption === true
        ) {
            this.transition = new ScreenTransition();
        } else if (isPlainObject(transitionOption)) {
            this.transition = new ScreenTransition(transitionOption);
        } else {
            throw new TypeError(
                'ScreenHost option "transition" must be a boolean or a plain object.'
            );
        }

        this.container = null;
        this._currentElement = null;
        this._currentScrollKey = null;
        this._scrollPositions = new Map(); // entryId -> scrollTop
        this._listener = event => this._renderCurrent(this.container, event);
    }

    get isMounted() {
        return this.container !== null;
    }

    mount(container) {
        if (!(container instanceof Element)) {
            throw new TypeError("ScreenHost requires a valid DOM container.");
        }

        if (this.container) {
            throw new Error("ScreenHost is already mounted. Unmount it first.");
        }

        // Render first so a rendering failure doesn't leave
        // a partially mounted host or an orphaned listener.
        this._renderCurrent(container, null);

        this.container = container;
        container.classList.add("screen-host");

        this.navigation.on("change", this._listener);

        return this;
    }

    unmount() {
        if (!this.container) return this;

        this.navigation.off("change", this._listener);
        this.transition?.destroy();
        this.container.replaceChildren();
        this.container.classList.remove("screen-host");

        // Don't leak navigation state onto a re-usable container.
        delete this.container.dataset.canGoBack;
        delete this.container.dataset.canGoForward;
            
        this.container = null;
        this._currentElement = null;
        this._currentScrollKey = null;
        this._scrollPositions.clear();

        return this;
    }

    _getCurrentOutput() {
        const current = this.navigation.current;

        if (!current) {
            throw new Error(
                "ScreenHost cannot render because no screen is active."
            );
        }

        // TabNavigator.current wraps its screen snapshot;
        // ScreenManager.current is already the screen snapshot.
        const screen = current.screen ?? current;

        if (!screen || screen.output === undefined) {
            throw new Error("The active screen has no render output.");
        }

        return screen.output;
    }

    // add above _renderCurrent
    _scrollKey() {
        const current = this.navigation.current;
        if (!current) return null;

        const entryId = current.screen?.id ?? current.id;

        if (entryId === undefined || entryId === null) return null;

        const tabId = current.tabId;

        return tabId !== undefined ? `${tabId}:${entryId}` : `_:${entryId}`;
    }

    _renderCurrent(container = this.container, event = null) {
        if (!container) return null;

        // 1. Capture from the outgoing screen, keyed by tab + entry.
        if (
            this.preserveScroll &&
            this._currentScrollKey !== null &&
            this._currentElement
        ) {
            const scrollTop = this._currentElement.scrollTop;

            if (scrollTop > 0) {
                this._scrollPositions.set(this._currentScrollKey, scrollTop);
            } else {
                this._scrollPositions.delete(this._currentScrollKey);
            }
        }

        // 2. Resolve the incoming output and key.
        const output = this._getCurrentOutput();

        const newScrollKey = this._scrollKey();

        const incoming = this.renderer.renderToElement(output);
        const outgoing = this._currentElement;

        this._currentElement = incoming;
        this._currentScrollKey = newScrollKey;

        incoming.classList.add("screen-host__screen");

        // 3. Transition or hard swap.
        if (this.transition && outgoing && event) {
            this.transition.play({
                container,
                outgoing,
                incoming,
                direction: directionFor(event.action)
            });
        } else {
            container.replaceChildren(incoming);
        }

        // 4. Restore — only for the same tab + entry.
        if (this.preserveScroll && newScrollKey !== null) {
            const saved = this._scrollPositions.get(newScrollKey);

            if (saved !== undefined) {
                incoming.scrollTop = saved;
            }
        }

        this._updateNavigationAttributes(container);

        return incoming;
    }

_updateNavigationAttributes(container = this.container) {
    if (!container) return;

    const canGoBack =
        typeof this.navigation.canGoBack === "function"
            ? this.navigation.canGoBack()
            : false;

    const canGoForward =
        typeof this.navigation.canGoForward === "function"
            ? this.navigation.canGoForward()
            : false;

    container.dataset.canGoBack = String(canGoBack);
    container.dataset.canGoForward = String(canGoForward);
}
    
}

function isPlainObject(value) {
    if (value === null || typeof value !== "object") {
        return false;
    }

    const prototype = Object.getPrototypeOf(value);

    return prototype === Object.prototype || prototype === null;
}

function normalizeParams(params) {
    if (params === undefined) {
        return Object.freeze({});
    }

    if (!isPlainObject(params)) {
        throw new TypeError("Screen parameters must be a plain object.");
    }

    return Object.freeze({ ...params });
}

export default class ScreenManager {
    constructor(registry) {
        if (!registry || typeof registry.require !== "function") {
            throw new TypeError("ScreenManager requires a ScreenRegistry.");
        }

        this.registry = registry;
        this._stack = [];
        this._forward = [];

        this._listeners = new Map();
        this._nextEntryId = 1;
        this._suspended = false;
    }

    get current() {
        const entry = this._getCurrentEntry();

        return entry ? this._createSnapshot(entry) : null;
    }

    get stack() {
        return Object.freeze(
            this._stack.map(entry => this._createSnapshot(entry))
        );
    }

    canGoBack() {
        return this._stack.length > 1;
    }

    canGoForward() {
        return this._forward.length > 0;
    }

    get forwardStack() {
        return Object.freeze(
            this._forward.map(entry => this._createSnapshot(entry))
        );
    }

    on(eventName, listener) {
        if (typeof eventName !== "string" || eventName.trim() === "") {
            throw new TypeError("Event name must be a non-empty string.");
        }

        if (typeof listener !== "function") {
            throw new TypeError("Event listener must be a function.");
        }

        if (!this._listeners.has(eventName)) {
            this._listeners.set(eventName, new Set());
        }

        this._listeners.get(eventName).add(listener);

        return this;
    }

    off(eventName, listener) {
        const listeners = this._listeners.get(eventName);

        if (listeners) {
            listeners.delete(listener);

            if (listeners.size === 0) {
                this._listeners.delete(eventName);
            }
        }

        return this;
    }

    navigate(name, params = {}) {
        this._assertAvailable();

        const entry = this._createEntry(name, params);
        const previous = this._getCurrentEntry();

        if (previous) {
            this._deactivateForTransition(previous);
        }

        this._stack.push(entry);
        this._disposeForwardStack();

        try {
            this._activate(entry);
        } catch (error) {
            this._stack.pop();
            if (previous) this._activate(previous);
            throw error;
        }

        this._emitNavigation("navigate", previous, entry);

        return this.current;
    }

    _disposeForwardStack() {
        for (const entry of this._forward) {
            this._disposeEntry(entry);
        }
        this._forward = [];
    }

    forward() {
        this._assertAvailable();

        if (!this.canGoForward()) {
            return false;
        }

        const leaving = this._getCurrentEntry();
        const next = this._forward[this._forward.length - 1];

        this._deactivateForTransition(leaving);
        this._forward.pop();

        this._stack.push(next);

        try {
            this._activate(next);
        } catch (error) {
            this._stack.pop();
            this._forward.push(next);
            this._activate(leaving);
            throw error;
        }

        this._emitNavigation("forward", leaving, next);

        return this.current;
    }

    back() {
        this._assertAvailable();

        if (!this.canGoBack()) {
            return false;
        }

        const leaving = this._getCurrentEntry();
        const previous = this._stack[this._stack.length - 2];

        this._deactivateForTransition(leaving);
        this._stack.pop();

        try {
            this._activate(previous);
        } catch (error) {
            this._stack.push(leaving);
            this._activate(leaving);
            throw error;
        }

        this._forward.push(leaving); // ← new — only after activation succeeds

        this._emitNavigation("back", leaving, previous);

        return this.current;
    }

    replace(name, params = {}) {
        this._assertAvailable();

        const entry = this._createEntry(name, params);
        const previous = this._getCurrentEntry();

        if (!previous) {
            this._stack.push(entry);
            this._disposeForwardStack();

            try {
                this._activate(entry);
            } catch (error) {
                this._stack.pop();
                throw error;
            }

            this._emitNavigation("replace", null, entry);

            return this.current;
        }

        this._deactivateForTransition(previous);
        this._stack.pop();
        this._stack.push(entry);
        this._disposeForwardStack();

        try {
            this._activate(entry);
        } catch (error) {
            this._stack.pop();
            this._stack.push(previous);
            this._activate(previous);

            throw error;
        }

        this._disposeEntry(previous);
        this._emitNavigation("replace", previous, entry);

        return this.current;
    }

    reset(name, params = {}) {
        this._assertAvailable();

        const entry = this._createEntry(name, params);
        const previousStack = [...this._stack];
        const previous = this._getCurrentEntry();

        if (previous) {
            this._deactivateForTransition(previous);
        }

        this._stack = [entry];
        this._disposeForwardStack();

        try {
            this._activate(entry);
        } catch (error) {
            this._stack = previousStack;

            if (previous) {
                this._activate(previous);
            }

            throw error;
        }

        for (const oldEntry of previousStack) {
            this._disposeEntry(oldEntry);
        }

        this._emitNavigation("reset", previous, entry);

        return this.current;
    }

    get isSuspended() {
        return this._suspended;
    }

    suspend() {
        if (this._suspended) {
            return false;
        }

        const current = this._getCurrentEntry();

        if (current) {
            this._deactivateForTransition(current);
        }

        this._suspended = true;
        return true;
    }

    resume() {
        if (!this._suspended) {
            return false;
        }

        const current = this._getCurrentEntry();

        if (current) {
            this._activate(current);
        }

        this._suspended = false;
        return true;
    }

    _assertAvailable() {
        if (this._suspended) {
            throw new Error(
                "Cannot navigate a suspended screen stack. " +
                    "Resume it first."
            );
        }
    }

    _createEntry(name, params) {
        const definition = this.registry.require(name);

        return {
            id: this._nextEntryId++,
            name,
            params: normalizeParams(params),
            definition,
            instance: null,
            output: undefined,
            rendered: false,
            active: false
        };
    }

    _getCurrentEntry() {
        return this._stack.length ? this._stack[this._stack.length - 1] : null;
    }

    _createContext(entry) {
        return Object.freeze({
            name: entry.name,
            params: entry.params,
            entryId: entry.id,
            navigation: this
        });
    }

    _activate(entry) {
        if (entry.active) {
            return;
        }

        const context = this._createContext(entry);

        entry.active = true;

        try {
            if (entry.definition.kind === "class") {
                if (!entry.instance) {
                    entry.instance = new entry.definition.ScreenClass(context);
                }

                entry.instance.onEnter();

                if (!entry.rendered) {
                    entry.output = entry.instance.render();
                    entry.rendered = true;
                }
            } else if (!entry.rendered) {
                entry.output = entry.definition.render(context);
                entry.rendered = true;
            }
        } catch (error) {
            entry.active = false;

            if (entry.instance) {
                try {
                    entry.instance.onDispose();
                } finally {
                    entry.instance = null;
                }
            }

            entry.output = undefined;
            entry.rendered = false;

            throw error;
        }
    }

    _deactivate(entry) {
        if (!entry || !entry.active) {
            return;
        }

        // If onLeave fails, the screen remains active.
        if (entry.instance) {
            entry.instance.onLeave();
        }

        entry.active = false;

        if (!entry.definition.options.preserveState) {
            this._disposeInstance(entry);
            entry.output = undefined;
            entry.rendered = false;
        }
    }

    _deactivateForTransition(entry) {
        try {
            this._deactivate(entry);
        } catch (error) {
            // A failed onLeave leaves the screen active.
            // A failed disposal may happen after deactivation,
            // so attempt to restore that screen.
            if (!entry.active) {
                try {
                    this._activate(entry);
                } catch (recoveryError) {
                    throw new AggregateError(
                        [error, recoveryError],
                        "Screen transition failed and the previous screen " +
                            "could not be restored."
                    );
                }
            }

            throw error;
        }
    }

    _disposeInstance(entry) {
        if (!entry.instance) {
            return;
        }

        const instance = entry.instance;

        entry.instance = null;

        try {
            instance.onDispose();
        } finally {
            entry.output = undefined;
            entry.rendered = false;
        }
    }

    _disposeEntry(entry) {
        if (!entry) {
            return;
        }

        this._disposeInstance(entry);

        entry.output = undefined;
        entry.rendered = false;
        entry.active = false;
    }

    _createSnapshot(entry) {
        return Object.freeze({
            id: entry.id,
            name: entry.name,
            params: entry.params,
            output: entry.output,
            instance: entry.instance,
            active: entry.active
        });
    }

    _emit(eventName, event) {
        const listeners = this._listeners.get(eventName);

        if (!listeners) {
            return;
        }

        for (const listener of [...listeners]) {
            listener(event);
        }
    }

    _emitNavigation(action, previous, current) {
        const event = Object.freeze({
            action,
            previous: previous ? this._createSnapshot(previous) : null,
            current: this._createSnapshot(current),
            stack: this.stack
        });

        this._emit(action, event);
        this._emit("change", event);
    }
}

const DEFAULT_DURATION = 300;
const FALLBACK_GRACE = 50;
const DURATION_PROPERTY = "--screen-transition-duration";

const SCREEN_CLASS = "screen-host__screen";
const CURRENT_CLASS = "screen-host__screen--current";
const LEAVING_CLASS = "screen-host__screen--leaving";

const DIRECTIONS = new Set(["forward", "backward", "fade"]);

function isPlainObject(value) {
    if (value === null || typeof value !== "object") {
        return false;
    }
    const prototype = Object.getPrototypeOf(value);
    return prototype === Object.prototype || prototype === null;
}

export default class ScreenTransition {
    constructor(options = {}) {
        if (!isPlainObject(options)) {
            throw new TypeError("ScreenTransition options must be a plain object.");
        }

        for (const key of Object.keys(options)) {
            if (key !== "duration") {
                throw new TypeError(`Unknown ScreenTransition option "${key}".`);
            }
        }

        const duration = options.duration ?? DEFAULT_DURATION;

        if (
            typeof duration !== "number" ||
            !Number.isFinite(duration) ||
            duration < 0
        ) {
            throw new TypeError(
                "ScreenTransition duration must be a non-negative number."
            );
        }

        this.duration = duration;
        this._active = null;

        this._mediaQuery =
            typeof window !== "undefined" &&
            typeof window.matchMedia === "function"
                ? window.matchMedia("(prefers-reduced-motion: reduce)")
                : null;
    }

    get isAnimating() {
        return this._active !== null;
    }

    play({ container, outgoing, incoming, direction = "fade", onComplete } = {}) {
        if (!(container instanceof Element)) {
            throw new TypeError(
                "ScreenTransition.play() requires a DOM container."
            );
        }

        if (!(incoming instanceof Element)) {
            throw new TypeError(
                "ScreenTransition.play() requires an incoming element."
            );
        }

        if (
            outgoing !== undefined &&
            outgoing !== null &&
            !(outgoing instanceof Element)
        ) {
            throw new TypeError(
                "ScreenTransition.play() outgoing must be an Element or null."
            );
        }

        if (!DIRECTIONS.has(direction)) {
            throw new TypeError(
                `Unknown ScreenTransition direction "${direction}".`
            );
        }

        // An in-flight transition is force-completed rather than queued.
        // A second tap should feel responsive, not sluggish.
        if (this._active) {
            try {
                this._finish();
            } catch (error) {
                console.error(
                    "ScreenTransition: previous onComplete threw during interrupt",
                    error
                );
            }
        }

        const canAnimate =
            outgoing instanceof Element &&
            outgoing !== incoming &&
            outgoing.parentNode === container &&
            !this._prefersReducedMotion() &&
            this.duration > 0;

        if (!canAnimate) {
            if (
                outgoing instanceof Element &&
                outgoing !== incoming &&
                outgoing.parentNode
            ) {
                outgoing.remove();
            }
            if (!incoming.isConnected) {
                container.appendChild(incoming);
            }
            incoming.classList.add(SCREEN_CLASS);
            onComplete?.();
            return;
        }

        const directionToken = `screen-host--${direction}`;

        container.style.setProperty(DURATION_PROPERTY, `${this.duration}ms`);
        container.classList.add(directionToken);

        incoming.classList.add(SCREEN_CLASS, CURRENT_CLASS);
        outgoing.classList.add(SCREEN_CLASS, LEAVING_CLASS);

        container.appendChild(incoming);

        // Force a synchronous style flush so the browser observes the
        // "from" state of the animation before the next paint.
        void incoming.getBoundingClientRect();

        let finished = false;

        const finish = () => {
            if (finished) return;
            finished = true;
            this._finish();
        };

        incoming.addEventListener("animationend", finish);

        const timer = setTimeout(finish, this.duration + FALLBACK_GRACE);

        this._active = {
            container,
            outgoing,
            incoming,
            directionToken,
            onComplete,
            cleanup: () => {
                clearTimeout(timer);
                incoming.removeEventListener("animationend", finish);
            }
        };
    }

    cancel() {
        this._finish();
    }

    destroy() {
        this._finish();
    }

    _prefersReducedMotion() {
        return this._mediaQuery?.matches === true;
    }

    _finish() {
        const active = this._active;

        if (!active) {
            return;
        }

        this._active = null;

        const {
            cleanup,
            onComplete,
            container,
            outgoing,
            incoming,
            directionToken
        } = active;

        cleanup();

        if (outgoing && outgoing.parentNode === container) {
            outgoing.remove();
        }

        incoming.classList.remove(CURRENT_CLASS, LEAVING_CLASS);

        if (outgoing) {
            outgoing.classList.remove(CURRENT_CLASS, LEAVING_CLASS);
        }

        container.classList.remove(directionToken);

        onComplete?.();
    }
}

const DEFAULT_MIN_DISTANCE = 60;
const DEFAULT_MIN_VELOCITY = 0.3;
const DEFAULT_DIRECTION_RATIO = 1.5;
const CLAIM_DISTANCE = 10;

function isPlainObject(value) {
    if (value === null || typeof value !== "object") return false;
    const prototype = Object.getPrototypeOf(value);
    return prototype === Object.prototype || prototype === null;
}

export default class SwipeNavigation {
    constructor(navigator, options = {}) {
        if (
            !navigator ||
            typeof navigator.selectTab !== "function" ||
            !Array.isArray(navigator.tabs)
        ) {
            throw new TypeError("SwipeNavigation requires a TabNavigator.");
        }

        if (!isPlainObject(options)) {
            throw new TypeError(
                "SwipeNavigation options must be a plain object."
            );
        }

        const allowed = new Set([
            "minDistance",
            "minVelocity",
            "directionRatio",
            "wrap",
            "ignoreSelector"
        ]);

        for (const key of Object.keys(options)) {
            if (!allowed.has(key)) {
                throw new TypeError(`Unknown SwipeNavigation option "${key}".`);
            }
        }

        const minDistance = options.minDistance ?? DEFAULT_MIN_DISTANCE;
        if (typeof minDistance !== "number" || minDistance < 0) {
            throw new TypeError(
                "SwipeNavigation minDistance must be a non-negative number."
            );
        }

        const minVelocity = options.minVelocity ?? DEFAULT_MIN_VELOCITY;
        if (typeof minVelocity !== "number" || minVelocity < 0) {
            throw new TypeError(
                "SwipeNavigation minVelocity must be a non-negative number."
            );
        }

        const directionRatio =
            options.directionRatio ?? DEFAULT_DIRECTION_RATIO;
        if (typeof directionRatio !== "number" || directionRatio < 1) {
            throw new TypeError(
                "SwipeNavigation directionRatio must be a number >= 1."
            );
        }

        if (options.wrap !== undefined && typeof options.wrap !== "boolean") {
            throw new TypeError("SwipeNavigation wrap must be a boolean.");
        }

        if (
            options.ignoreSelector !== undefined &&
            typeof options.ignoreSelector !== "string"
        ) {
            throw new TypeError(
                "SwipeNavigation ignoreSelector must be a string."
            );
        }

        this.navigator = navigator;
        this.minDistance = minDistance;
        this.minVelocity = minVelocity;
        this.directionRatio = directionRatio;
        this.wrap = options.wrap ?? false;
        this.ignoreSelector = options.ignoreSelector ?? null;

        this.container = null;
        this._pointer = null;

        this._onPointerDown = event => this._handlePointerDown(event);
        this._onPointerMove = event => this._handlePointerMove(event);
        this._onPointerUp = event => this._handlePointerUp(event);
        this._onPointerCancel = event => this._handlePointerCancel(event);
    }

    get isMounted() {
        return this.container !== null;
    }

    mount(container) {
        if (!(container instanceof Element)) {
            throw new TypeError(
                "SwipeNavigation requires a valid DOM container."
            );
        }

        if (this.container) {
            throw new Error(
                "SwipeNavigation is already mounted. Unmount it first."
            );
        }

        this.container = container;

        container.addEventListener("pointerdown", this._onPointerDown);
        // Move/up on document so the gesture survives
        // the pointer leaving the container mid-swipe.
        document.addEventListener("pointermove", this._onPointerMove);
        document.addEventListener("pointerup", this._onPointerUp);
        document.addEventListener("pointercancel", this._onPointerCancel);

        return this;
    }

    unmount() {
        if (!this.container) return this;

        this.container.removeEventListener("pointerdown", this._onPointerDown);
        document.removeEventListener("pointermove", this._onPointerMove);
        document.removeEventListener("pointerup", this._onPointerUp);
        document.removeEventListener("pointercancel", this._onPointerCancel);

        this.container = null;
        this._pointer = null;

        return this;
    }

    // Testability seam — tests override this instance method.
    _now() {
        return performance.now();
    }

    _handlePointerDown(event) {
        if (event.isPrimary === false) return;
        if (this._pointer !== null) return;

        if (this.ignoreSelector) {
            const target =
                event.target instanceof Element ? event.target : null;

            if (target?.closest(this.ignoreSelector)) return;
        }

        this._pointer = {
            id: event.pointerId ?? 0,
            startX: event.clientX,
            startY: event.clientY,
            startTime: this._now(),
            claimed: false
        };
    }

    _handlePointerMove(event) {
        if (!this._pointer) return;
        if ((event.pointerId ?? 0) !== this._pointer.id) return;

        const dx = event.clientX - this._pointer.startX;
        const dy = event.clientY - this._pointer.startY;

        if (this._pointer.claimed) return;

        if (Math.hypot(dx, dy) < CLAIM_DISTANCE) return;

        const horizontalWins =
            Math.abs(dx) >= Math.abs(dy) * this.directionRatio;

        if (horizontalWins) {
            this._pointer.claimed = true;
        } else {
            // Vertical dominates — let the browser have it.
            this._pointer = null;
        }
    }

    _handlePointerUp(event) {
        if (!this._pointer) return;
        if ((event.pointerId ?? 0) !== this._pointer.id) return;

        const pointer = this._pointer;
        this._pointer = null;

        if (!pointer.claimed) return;

        const dx = event.clientX - pointer.startX;
        if (dx === 0) return;

        const duration = Math.max(1, this._now() - pointer.startTime);
        const velocity = Math.abs(dx) / duration;

        const meetsThreshold =
            Math.abs(dx) >= this.minDistance || velocity >= this.minVelocity;

        if (!meetsThreshold) return;

        // Paging only at tab root.
        if (this.navigator.stack.length !== 1) return;

        const tabs = this.navigator.tabs;
        if (tabs.length < 2) return;

        const currentIndex = tabs.findIndex(
            tab => tab.id === this.navigator.selectedTabId
        );
        if (currentIndex < 0) return;

        // Swipe left (dx < 0) → next tab.
        // Swipe right (dx > 0) → previous tab.
        const targetIndex = currentIndex - Math.sign(dx);

        let resolved = targetIndex;

        if (this.wrap) {
            resolved = (targetIndex + tabs.length) % tabs.length;
        } else if (targetIndex < 0 || targetIndex >= tabs.length) {
            return;
        }

        this.navigator.selectTab(tabs[resolved].id);
    }

    _handlePointerCancel(event) {
        if (!this._pointer) return;
        if ((event.pointerId ?? 0) !== this._pointer.id) return;
        this._pointer = null;
    }
}

function isPlainObject(value) {

function validateTabs(tabs) {
    if (!Array.isArray(tabs) || tabs.length === 0) {
        throw new TypeError("TabNavigator requires a non-empty array of tabs.");
    }

    const ids = new Set();

    return tabs.map((tab, index) => {
        if (!isPlainObject(tab)) {
            throw new TypeError(
                `Tab at index ${index} must be a plain object.`
            );
        }

        for (const key of Object.keys(tab)) {
            if (!["id", "label", "icon", "screen"].includes(key)) {
                throw new TypeError(`Unknown tab property "${key}".`);
            }
        }

        if (
            typeof tab.id !== "string" ||
            !/^[A-Za-z][A-Za-z0-9_-]*$/.test(tab.id)
        ) {
            throw new TypeError(`Invalid tab id "${tab.id}".`);
        }

        if (ids.has(tab.id)) {
            throw new Error(`Tab "${tab.id}" is already defined.`);
        }

        ids.add(tab.id);

        if (typeof tab.label !== "string" || tab.label.trim() === "") {
            throw new TypeError(`Tab "${tab.id}" requires a non-empty label.`);
        }

        if (
            tab.icon !== undefined &&
            (typeof tab.icon !== "string" || tab.icon.trim() === "")
        ) {
            throw new TypeError(
                `Tab "${tab.id}" icon must be a non-empty string.`
            );
        }

        if (typeof tab.screen !== "string" || tab.screen.trim() === "") {
            throw new TypeError(
                `Tab "${tab.id}" requires an initial screen name.`
            );
        }

        return Object.freeze({
            id: tab.id,
            label: tab.label,
            icon: tab.icon ?? null,
            screen: tab.screen
        });
    });
}

export default class TabNavigator {
    constructor(registry, tabs) {
        if (!registry || typeof registry.require !== "function") {
            throw new TypeError("TabNavigator requires a ScreenRegistry.");
        }

        this.registry = registry;
        this._tabs = validateTabs(tabs);
        this._tabById = new Map();
        this._managers = new Map();
        this._listeners = new Map();
        this._selectedTabId = null;

        for (const tab of this._tabs) {
            // Fail early if a tab references an unknown screen.
            this.registry.require(tab.screen);

            this._tabById.set(tab.id, tab);
            this._managers.set(tab.id, new ScreenManager(this.registry));
        }

        this.selectTab(this._tabs[0].id);
    }

    get selectedTabId() {
        return this._selectedTabId;
    }

    get current() {
        const manager = this._getCurrentManager();

        if (!manager || !manager.current) {
            return null;
        }

        return Object.freeze({
            tabId: this._selectedTabId,
            tab: this._tabById.get(this._selectedTabId),
            screen: manager.current
        });
    }

    get tabs() {
        return Object.freeze(
            this._tabs.map(tab =>
                Object.freeze({
                    ...tab,
                    selected: tab.id === this._selectedTabId,
                    initialized: this._managers.get(tab.id).stack.length > 0,
                    stackLength: this._managers.get(tab.id).stack.length
                })
            )
        );
    }

    get stack() {
        const manager = this._getCurrentManager();
        return manager ? manager.stack : Object.freeze([]);
    }

    get forwardStack() {
        // ← new
        const manager = this._getCurrentManager();
        return manager ? manager.forwardStack : Object.freeze([]);
    }

    canGoBack() {
        const manager = this._getCurrentManager();
        return manager ? manager.canGoBack() : false;
    }

    canGoForward() {
        // ← new
        const manager = this._getCurrentManager();
        return manager ? manager.canGoForward() : false;
    }

    selectTab(id, params = {}) {
        const tab = this._tabById.get(id);

        if (!tab) {
            throw new Error(`Unknown tab "${id}".`);
        }

        if (id === this._selectedTabId) {
            return this.current;
        }

        const previousId = this._selectedTabId;
        const previousManager = previousId
            ? this._managers.get(previousId)
            : null;
        const nextManager = this._managers.get(id);

        if (previousManager) {
            previousManager.suspend();
        }

        try {
            if (nextManager.stack.length === 0) {
                nextManager.navigate(tab.screen, params);
            } else {
                nextManager.resume();
            }
        } catch (error) {
            if (previousManager) {
                try {
                    previousManager.resume();
                } catch (recoveryError) {
                    throw new AggregateError(
                        [error, recoveryError],
                        "Tab selection failed and the previous tab " +
                            "could not be resumed."
                    );
                }
            }

            throw error;
        }

        this._selectedTabId = id;

        this._emit("tabChange", this._createEvent("selectTab", previousId, id));

        this._emit("change", this._createEvent("selectTab", previousId, id));

        return this.current;
    }

    navigate(name, params = {}) {
        const manager = this._requireCurrentManager();
        const result = manager.navigate(name, params);

        this._emit(
            "change",
            this._createEvent(
                "navigate",
                this._selectedTabId,
                this._selectedTabId
            )
        );

        return result;
    }

    back() {
        const manager = this._requireCurrentManager();
        const result = manager.back();
        if (result === false) return false;

        this._emit(
            "change",
            this._createEvent("back", this._selectedTabId, this._selectedTabId)
        );

        return this.current;
    }

    forward() {
        // ← new
        const manager = this._requireCurrentManager();
        const result = manager.forward();

        if (result === false) {
            return false;
        }

        this._emit(
            "change",
            this._createEvent(
                "forward",
                this._selectedTabId,
                this._selectedTabId
            )
        );

        return this.current;
    }

    replace(name, params = {}) {
        const manager = this._requireCurrentManager();
        manager.replace(name, params);

        this._emit(
            "change",
            this._createEvent(
                "replace",
                this._selectedTabId,
                this._selectedTabId
            )
        );

        return this.current;
    }

    reset(name, params = {}) {
        const manager = this._requireCurrentManager();
        manager.reset(name, params);

        this._emit(
            "change",
            this._createEvent("reset", this._selectedTabId, this._selectedTabId)
        );

        return this.current;
    }

    restore(tabId, entries, { silent = false } = {}) {
        const tab = this._tabById.get(tabId);

        if (!tab) {
            throw new Error(`Unknown tab "${tabId}".`);
        }

        if (!Array.isArray(entries) || entries.length === 0) {
            throw new TypeError(
                "restore() requires a non-empty array of entries."
            );
        }

        for (const entry of entries) {
            if (!entry || typeof entry.name !== "string") {
                throw new TypeError(
                    "restore() entries require a string 'name' property."
                );
            }
            this.registry.require(entry.name);
        }

        const manager = this._managers.get(tabId);
        const prevId = this._selectedTabId;

        // Capture BEFORE reset()/navigate() clear these.
        const prevStack = [...manager._stack];
        const prevForward = [...manager._forward];
        const prevDepth = prevStack.length;
        const targetDepth = entries.length;

        if (prevId && prevId !== tabId) {
            this._managers.get(prevId).suspend();
        }

        if (manager.isSuspended) {
            manager.resume();
        }

        manager.reset(entries[0].name, entries[0].params ?? {});

        for (let i = 1; i < entries.length; i++) {
            manager.navigate(entries[i].name, entries[i].params ?? {});
        }

        // Rebuild forward stack (reset() and navigate() just cleared it).
        // `_forward` LIFO: end of array is the next forward target.
        if (targetDepth < prevDepth) {
            // Went back — removed stack entries become the new forward top.
            // Reverse so the shallowest removed entry is popped first.
            const removed = prevStack.slice(targetDepth).reverse();
            manager._forward = [...prevForward, ...removed];
        } else if (targetDepth > prevDepth) {
            // Went forward — pop the added count from existing forward.
            const added = targetDepth - prevDepth;
            manager._forward = prevForward.slice(
                0,
                Math.max(0, prevForward.length - added)
            );
        } else {
            // Same depth — preserve existing forward.
            manager._forward = prevForward;
        }

        this._selectedTabId = tabId;

        if (!silent) {
            this._emit("change", this._createEvent("restore", prevId, tabId));
        }

        return this.current;
    }

    on(eventName, listener) {
        if (typeof eventName !== "string" || eventName.trim() === "") {
            throw new TypeError("Event name must be a non-empty string.");
        }

        if (typeof listener !== "function") {
            throw new TypeError("Event listener must be a function.");
        }

        if (!this._listeners.has(eventName)) {
            this._listeners.set(eventName, new Set());
        }

        this._listeners.get(eventName).add(listener);

        return this;
    }

    off(eventName, listener) {
        const listeners = this._listeners.get(eventName);

        if (listeners) {
            listeners.delete(listener);

            if (listeners.size === 0) {
                this._listeners.delete(eventName);
            }
        }

        return this;
    }

    _getCurrentManager() {
        return this._selectedTabId
            ? this._managers.get(this._selectedTabId)
            : null;
    }

    _requireCurrentManager() {
        const manager = this._getCurrentManager();

        if (!manager) {
            throw new Error("No tab is currently selected.");
        }

        return manager;
    }

    _createEvent(action, previousTabId, tabId) {
        const manager = this._getCurrentManager();

        return Object.freeze({
            action,
            previousTabId,
            tabId,
            canGoBack: manager ? manager.canGoBack() : false,
            canGoForward: manager ? manager.canGoForward() : false,
            current: this.current,
            tabs: this.tabs,
            stack: this.stack
        });
    }

    _emit(eventName, event) {
        const listeners = this._listeners.get(eventName);

        if (!listeners) {
            return;
        }

        for (const listener of [...listeners]) {
            listener(event);
        }
    }
}

import { normalizeLayoutProps } from "../../core/Layout.js";
import normalizeComponent from "../../core/normalizeComponent.js";

export default class DOMRenderer {
    renderToElement(component) {
        const normalized = normalizeComponent(component);
        return this.createElement(normalized);
    }
    
    render(component, container) {
        if (!(container instanceof Element)) {
            throw new TypeError("DOMRenderer requires a valid DOM container.");
        }
        const element = this.renderToElement(component);
        container.replaceChildren(element);
        return element;
    }

    createElement(component) {
        if (typeof component === "string" || typeof component === "number") {
            return document.createTextNode(String(component));
        }

        const props = component.props ?? {};
        const type = component.type;

        if (type === "fragment") {
            const fragment = document.createDocumentFragment();

            for (const child of component.children ?? []) {
                fragment.appendChild(this.createElement(child));
            }

            return fragment;
        }

        const isLayout = type === "column" || type === "row";

        const tagName =
            type === "button"
                ? "button"
                : (props.tagName ??
                  (type === "card"
                      ? "article"
                      : type === "text"
                        ? "p"
                        : "div"));

        const element = document.createElement(tagName);

        if (isLayout || type === "card" || type === "button") {
            element.classList.add(`ui-${type}`);
        }

        if (type === "button") {
            element.classList.add(
                `ui-button-${props.variant}`,
                `ui-button-${props.size}`
            );

            element.disabled = Boolean(props.disabled);
            element.type = props.type;
        }

        if (
            isLayout ||
            type === "card" ||
            type === "button" ||
            type === "text"
        ) {
            element.classList.add(`ui-${type}`);
        }

        if (type === "button") {
            element.classList.add(
                `ui-button-${props.variant}`,
                `ui-button-${props.size}`
            );

            element.disabled = Boolean(props.disabled);
            element.type = props.type;
        }

        if (type === "text") {
            element.classList.add(`ui-text-${props.size}`);

            if (props.color) {
                element.style.color = props.color;
            }

            if (props.weight != null) {
                element.style.fontWeight = props.weight;
            }

            if (props.align) {
                element.style.textAlign = props.align;
            }
        }

        if (props.className) {
            element.classList.add(...props.className.trim().split(/\s+/));
        }

        this.applyLayout(element, props);

        if (props.attributes) {
            for (const [name, value] of Object.entries(props.attributes)) {
                if (value == null || value === false) {
                    continue;
                }

                element.setAttribute(name, value === true ? "" : String(value));
            }
        }

        if (typeof props.onClick === "function") {
            element.addEventListener("click", props.onClick);
        }

        for (const child of component.children ?? []) {
            element.appendChild(this.createElement(child));
        }

        return element;
    }

    applyLayout(element, props) {
        const layoutProps = element.classList.contains("ui-text")
            ? { ...props, align: undefined }
            : props;

        const layout = normalizeLayoutProps(layoutProps);
        const style = element.style;

        const alignValues = {
            start: "flex-start",
            center: "center",
            end: "flex-end",
            stretch: "stretch"
        };

        const justifyValues = {
            start: "flex-start",
            center: "center",
            end: "flex-end",
            between: "space-between",
            around: "space-around",
            evenly: "space-evenly"
        };

        for (const name of ["gap", "padding", "width", "height"]) {
            if (layout[name] != null) {
                style[name] = this.toCssSize(layout[name]);
            }
        }

        if (layout.align) {
            style.alignItems = alignValues[layout.align];
        }

        if (layout.justify) {
            style.justifyContent = justifyValues[layout.justify];
        }

        // Explicit custom styles override generated layout styles.
        if (props.style) {
            Object.assign(style, props.style);
        }

        if (props.wrap != null) {
            element.style.flexWrap =
                props.wrap === true
                    ? "wrap"
                    : props.wrap === false
                        ? "nowrap"
                        : props.wrap;
        }
    }

    toCssSize(value) {
        return typeof value === "number" ? `${value}px` : value;
    }
}
