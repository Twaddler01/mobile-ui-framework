import validateComponent from "./validateComponent.js";

function isPlainObject(value) {
    if (value === null || typeof value !== "object") {
        return false;
    }

    const prototype = Object.getPrototypeOf(value);

    return prototype === Object.prototype || prototype === null;
}

export default function validateDefinition(definition) {
    const errors = [];
    const activePath = new WeakSet();

    function visit(node, path) {
        if (typeof node === "string") {
            return;
        }

        if (typeof node === "number") {
            if (!Number.isFinite(node)) {
                errors.push(
                    `${path}: Numeric children must be finite numbers.`
                );
            }

            return;
        }

        if (node === null || typeof node !== "object") {
            errors.push(`${path}: Invalid component or child.`);
            return;
        }

        if (activePath.has(node)) {
            errors.push(
                `${path}: Circular component references are not allowed.`
            );
            return;
        }

        activePath.add(node);

        try {
            if (Array.isArray(node)) {
                node.forEach((child, index) => {
                    visit(child, `${path}[${index}]`);
                });

                return;
            }

            if (
                !isPlainObject(node) &&
                !(
                    typeof node.type === "string" &&
                    "props" in node &&
                    "children" in node
                )
            ) {
                errors.push(
                    `${path}: Expected a plain-object definition or component instance.`
                );

                return;
            }

            if (typeof node.type !== "string") {
                errors.push(`${path}: Component type must be a string.`);

                return;
            }

            const props = node.props ?? {};
            const children = node.children ?? [];

            errors.push(...validateComponent(node.type, props, path));

            if (!Array.isArray(children)) {
                errors.push(`${path}: Component children must be an array.`);

                return;
            }

            children.forEach((child, index) => {
                visit(child, `${path}.children[${index}]`);
            });
        } finally {
            activePath.delete(node);
        }
    }

    // A root must be a component definition or an array of nodes.
    if (definition === null || typeof definition !== "object") {
        return {
            valid: false,
            errors: ["root: The UI root must be a component or array."]
        };
    }

    visit(definition, "root");

    return {
        valid: errors.length === 0,
        errors
    };
}


========== src/core/normalizeComponent.js ==========

import { getComponentSchema } from "./componentSchemas.js";
import validateDefinition from "./validateDefinition.js";

function normalizeNode(node) {
    if (typeof node === "string" || typeof node === "number") {
        return node;
    }

    if (Array.isArray(node)) {
        return {
            type: "fragment",
            props: {},
            children: node.map(normalizeNode)
        };
    }

    const schema = getComponentSchema(node.type);

    if (!schema) {
        throw new TypeError(
            `Unknown component type "${node.type}".`
        );
    }

    const props = {
        ...schema.defaults,
        ...node.props
    };

    const children = (node.children ?? []).map(normalizeNode);

    return {
        type: node.type,
        props,
        children
    };
}

export default function normalizeComponent(component) {
    const result = validateDefinition(component);

    if (!result.valid) {
        throw new TypeError(
            `UI definition validation failed:\n- ${result.errors.join("\n- ")}`
        );
    }

    return normalizeNode(component);
}

import { normalizeLayoutProps } from "../../core/Layout.js";
import normalizeComponent from "../../core/normalizeComponent.js";

export default class DOMRenderer {
    render(component, container) {
        if (!(container instanceof Element)) {
            throw new TypeError("DOMRenderer requires a valid DOM container.");
        }

        const normalized = normalizeComponent(component);
        const element = this.createElement(normalized);

        container.replaceChildren(element);

        return element;
    }

    createElement(component) {
        if (typeof component === "string" || typeof component === "number") {
            return document.createTextNode(String(component));
        }

        const props = component.props ?? {};
        const type = component.type;

        if (type === "fragment") {
            const fragment = document.createDocumentFragment();

            for (const child of component.children ?? []) {
                fragment.appendChild(this.createElement(child));
            }

            return fragment;
        }

        const isLayout = type === "column" || type === "row";

        const tagName =
            type === "button"
                ? "button"
                : (props.tagName ??
                  (type === "card"
                      ? "article"
                      : type === "text"
                        ? "p"
                        : "div"));

        const element = document.createElement(tagName);

        if (isLayout || type === "card" || type === "button") {
            element.classList.add(`ui-${type}`);
        }

        if (type === "button") {
            element.classList.add(
                `ui-button-${props.variant}`,
                `ui-button-${props.size}`
            );

            element.disabled = Boolean(props.disabled);
            element.type = props.type;
        }

        if (
            isLayout ||
            type === "card" ||
            type === "button" ||
            type === "text"
        ) {
            element.classList.add(`ui-${type}`);
        }

        if (type === "button") {
            element.classList.add(
                `ui-button-${props.variant}`,
                `ui-button-${props.size}`
            );

            element.disabled = Boolean(props.disabled);
            element.type = props.type;
        }

        if (type === "text") {
            element.classList.add(`ui-text-${props.size}`);

            if (props.color) {
                element.style.color = props.color;
            }

            if (props.weight != null) {
                element.style.fontWeight = props.weight;
            }

            if (props.align) {
                element.style.textAlign = props.align;
            }
        }

        if (props.className) {
            element.classList.add(...props.className.trim().split(/\s+/));
        }

        this.applyLayout(element, props);

        if (props.attributes) {
            for (const [name, value] of Object.entries(props.attributes)) {
                if (value == null || value === false) {
                    continue;
                }

                element.setAttribute(name, value === true ? "" : String(value));
            }
        }

        if (typeof props.onClick === "function") {
            element.addEventListener("click", props.onClick);
        }

        for (const child of component.children ?? []) {
            element.appendChild(this.createElement(child));
        }

        return element;
    }

    applyLayout(element, props) {
        const layoutProps = element.classList.contains("ui-text")
            ? { ...props, align: undefined }
            : props;

        const layout = normalizeLayoutProps(layoutProps);
        const style = element.style;

        const alignValues = {
            start: "flex-start",
            center: "center",
            end: "flex-end",
            stretch: "stretch"
        };

        const justifyValues = {
            start: "flex-start",
            center: "center",
            end: "flex-end",
            between: "space-between",
            around: "space-around",
            evenly: "space-evenly"
        };

        for (const name of ["gap", "padding", "width", "height"]) {
            if (layout[name] != null) {
                style[name] = this.toCssSize(layout[name]);
            }
        }

        if (layout.align) {
            style.alignItems = alignValues[layout.align];
        }

        if (layout.justify) {
            style.justifyContent = justifyValues[layout.justify];
        }

        // Explicit custom styles override generated layout styles.
        if (props.style) {
            Object.assign(style, props.style);
        }

        if (props.wrap != null) {
            element.style.flexWrap =
                props.wrap === true
                    ? "wrap"
                    : props.wrap === false
                        ? "nowrap"
                        : props.wrap;
        }
    }

    toCssSize(value) {
        return typeof value === "number" ? `${value}px` : value;
    }
}


========== tests/componentSchemas.test.js ==========

import test from "node:test";
import assert from "node:assert/strict";

import validateComponent from "../src/core/validateComponent.js";
import {
    Column,
    Text,
    Button,
    validateDefinition,
    normalizeComponent,
    registerComponentSchema,
    getComponentSchema,
    hasComponentSchema,
    listComponentTypes
} from "../src/ui.js";

test("built-in schemas are registered", () => {
    assert.equal(hasComponentSchema("column"), true);
    assert.equal(hasComponentSchema("text"), true);
    assert.equal(hasComponentSchema("button"), true);
});

test("valid component trees pass validation", () => {
    const view = Column({ gap: 12, padding: 20 }, [
        Text({ size: "title" }, ["Welcome"]),
        Button({ variant: "primary" }, ["Continue"])
    ]);

    const result = validateDefinition(view);

    assert.equal(result.valid, true);
    assert.deepEqual(result.errors, []);
});

test("invalid properties are reported", () => {
    const result = validateDefinition({
        type: "button",
        props: {
            variant: "mystery",
            disabled: "no"
        },
        children: ["Test"]
    });

    assert.equal(result.valid, false);
    assert.ok(result.errors.some(error => error.includes("variant")));
    assert.ok(result.errors.some(error => error.includes("disabled")));
});

test("normalization applies defaults", () => {
    const normalized = normalizeComponent({
        type: "button",
        props: {},
        children: ["Continue"]
    });

    assert.equal(normalized.props.variant, "primary");
    assert.equal(normalized.props.size, "medium");
    assert.equal(normalized.props.disabled, false);
    assert.equal(normalized.props.type, "button");
});

test("invalid definitions cannot be normalized", () => {
    assert.throws(
        () =>
            normalizeComponent({
                type: "column",
                props: { gap: -1 },
                children: []
            }),
        /validation failed/i
    );
});

test("unknown properties are rejected", () => {
    const result = validateDefinition({
        type: "text",
        props: { colour: "red" },
        children: ["Hello"]
    });

    assert.equal(result.valid, false);
    assert.ok(result.errors.some(error => error.includes("colour")));
});

test("custom component schemas can be registered", () => {
    registerComponentSchema("test-label", {
        properties: {
            tone: {
                kind: "enum",
                values: ["neutral", "success", "warning"]
            }
        },
        defaults: {
            tone: "neutral"
        }
    });

    assert.equal(hasComponentSchema("test-label"), true);
    assert.equal(getComponentSchema("test-label").defaults.tone, "neutral");

    assert.ok(listComponentTypes().includes("test-label"));

    const result = validateDefinition({
        type: "test-label",
        props: { tone: "success" },
        children: ["Ready"]
    });

    assert.equal(result.valid, true);

    const normalized = normalizeComponent({
        type: "test-label",
        props: {},
        children: ["Ready"]
    });

    assert.equal(normalized.props.tone, "neutral");
});

test("duplicate registration is rejected by default", () => {
    assert.throws(
        () =>
            registerComponentSchema("button", {
                properties: {},
                defaults: {}
            }),
        /already registered/i
    );
});

test("malformed schema definitions are rejected", () => {
    assert.throws(
        () =>
            registerComponentSchema("bad-test", {
                properties: {
                    tone: { kind: "made-up-rule" }
                },
                defaults: {}
            }),
        /unsupported rule kind/i
    );
});

test("registered schemas and nested rules are immutable", () => {
    const schema = getComponentSchema("button");

    assert.ok(Object.isFrozen(schema));
    assert.ok(Object.isFrozen(schema.properties));
    assert.ok(Object.isFrozen(schema.properties.variant.values));
    assert.ok(Object.isFrozen(schema.defaults));

    assert.throws(() => {
        schema.defaults.variant = "ghost";
    }, TypeError);

    assert.throws(() => {
        schema.properties.variant.values.push("danger");
    }, TypeError);
});

test("required properties are enforced", () => {
    registerComponentSchema("test-required-field", {
        properties: {
            label: { kind: "string" }
        },
        required: ["label"]
    });

    assert.ok(
        validateComponent("test-required-field", {}).some(
            error => error.includes('Required property "label"')
        )
    );

    assert.deepEqual(
        validateComponent("test-required-field", {
            label: "Continue"
        }),
        []
    );
});

test("custom validators reject out-of-range numbers", () => {
    registerComponentSchema("test-progress-meter", {
        properties: {
            value: {
                kind: "number",
                validate: value =>
                    value >= 0 && value <= 100
                        ? true
                        : "Value must be between 0 and 100."
            }
        },
        required: ["value"]
    });

    assert.deepEqual(
        validateComponent("test-progress-meter", {
            value: 50
        }),
        []
    );

    assert.ok(
        validateComponent("test-progress-meter", {
            value: 120
        }).some(error => error.includes("between 0 and 100"))
    );
});

test("invalid custom validators are rejected during registration", () => {
    assert.throws(() => {
        registerComponentSchema("test-bad-validator", {
            properties: {
                value: {
                    kind: "number",
                    validate: "not a function"
                }
            }
        });
    }, TypeError);
});
